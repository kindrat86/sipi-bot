The 10 lines:

import urllib.request, json

def agent_may_spend(amount, merchant, category="saas"):
    req = urllib.request.Request(
        "https://sipi.bot/v1/transactions/evaluate",
        data=json.dumps({"amount": amount, "merchant": merchant, "category": category}).encode(),
        headers={"Content-Type": "application/json"})
    decision = json.load(urllib.request.urlopen(req))
    return decision.get("decision") == "APPROVED"

if not agent_may_spend(200, "openai-credits"):
    raise SystemExit("Blocked by spend firewall")
# ... only now let the agent actually buy
